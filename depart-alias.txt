    err("Use !depart drowned-watch")

    err("Set ledger_loadout first. This is a declared manifest, not imported D&D Beyond inventory.")

encoded = dump_json(data)
if len(encoded) > 3800:
    err("Export too large. Use fewer manifest entries; do not omit expedition-relevant gear.")
desc = "```json\n" + encoded + "\n```"
</drac2>
